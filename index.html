let context = { problem: null, address: null, account: null }; function detectIntent(text) { if (text.includes("вода")) return "water"; if (text.includes("отоплен")) return "heating"; if (text.includes("свет") || text.includes("электр")) return "electricity"; if (text.includes("плат") text.includes("деньг") text.includes("квитан")) return "billing"; if (text.includes("показан")) return "meters"; return "unknown"; } function handleSend() { const input = document.getElementById('userInput'); const rawText = input.value.trim(); const text = rawText.toLowerCase(); if (!text) return; addMessage(rawText, 'operator'); input.value = ''; // завершение диалога if ( text.includes("до свидания") || text.includes("всего доброго") || text.includes("спасибо") ) { addMessage("Благодарю за обращение. Хорошего дня!", 'client'); addMessage("Диалог завершен.", 'system'); isChatActive = false; return; } setTimeout(() => { switch (currentState) { case "start": const intent = detectIntent(text); if (intent === "unknown") { addMessage("Пожалуйста, уточните, у вас проблема с водой, отоплением, электричеством или начислениями?", 'client'); } else { context.problem = intent; currentState = "askAddress"; addMessage("Уточните, пожалуйста, адрес дома.", 'client'); } break; case "askAddress": context.address = rawText; currentState = "askAccount"; addMessage("Назовите, пожалуйста, номер лицевого счета.", 'client'); break; case "askAccount": context.account = rawText; currentState = "solution"; let solutionText = ""; if (context.problem === "water") { solutionText = "По вашему адресу возможны аварийные работы. Информация передана в службу."; } if (context.problem === "heating") { solutionText = "Зарегистрирована заявка на проверку отопления."; } if (context.problem === "electricity") { solutionText = "Передаю информацию в аварийную электрослужбу."; } if (context.problem === "billing") { solutionText = "Создан запрос на проверку начислений."; } addMessage(solutionText, 'client'); addMessage("Могу ли я вам чем-то еще помочь?", 'client'); currentState = "final"; break; case "final": addMessage("Если вопросов больше нет, благодарю за обращение.", 'client'); break; default: addMessage("Пожалуйста, уточните ваш вопрос.", 'client'); } }, 800); }
